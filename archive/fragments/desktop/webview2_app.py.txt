    def quit(self):
        """Закрыть программу из интерфейса (кнопка «Выйти» ведёт на экран входа, а не сюда)."""
        try:
            import webview
            for w in webview.windows:
                w.destroy()
            return {"ok": True}
        except Exception as e:      # noqa: BLE001
            return {"ok": False, "error": str(e)}
